-- 1. Delete classified entries (they are of no use, draft system is enough)
DELETE FROM public.projects WHERE classified = true;
DELETE FROM public.signals WHERE classified = true;

-- 2. Drop classified and old redaction columns (if they exist)
ALTER TABLE public.projects DROP COLUMN IF EXISTS classified;
ALTER TABLE public.signals DROP COLUMN IF EXISTS classified;
ALTER TABLE public.projects DROP COLUMN IF EXISTS redaction_blocks;
ALTER TABLE public.signals DROP COLUMN IF EXISTS redaction_blocks;
ALTER TABLE public.signals DROP COLUMN IF EXISTS date_redacted;

-- 3. Verify - should return 0 rows for classified columns
SELECT column_name, table_name 
FROM information_schema.columns 
WHERE table_name IN ('projects','signals') 
AND column_name IN ('classified','redaction_blocks','date_redacted');

-- 4. Verify counts - should be 4 projects, 2 signals published (or your current)
SELECT code, slug, status FROM public.projects ORDER BY code;
SELECT code, slug, status FROM public.signals ORDER BY code;
